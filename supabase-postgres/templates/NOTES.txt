# 1. Get the connection information:

     kubectl get --namespace {{ .Release.Namespace }} svc {{ include "supabase-postgres.fullname" . }} -o jsonpath="{.spec.clusterIP}"
     kubectl get --namespace {{ .Release.Namespace }} secret {{ .Release.Name }} -o jsonpath="{.data.password}" | base64 -d

# 2. Connect to the database:

     psql "host=<IP> port=5432 user=postgres password=<PASSWORD> dbname=postgres"
